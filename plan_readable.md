function onFormSubmit(e) {

      + "<b>Service:</b> " + serviceType + "<br>"

    // Send email to owner

      to: OWNER_EMAIL,
      subject: subject,
      htmlBody: htmlBody
    });

    // Optionally send SMS via email-to-sms gateway

      var smsText = "NEW LEAD: " + (name ? name : phone) + " | " + serviceType + " | " + phone;

      });
